// Independent review of e2e5066 / 42949e8 (HEAD 4535e07). Every test here states the SAFE behaviour
// ("not sure = leave the field blank / keep the line") and FAILS at HEAD: one failure = one finding.
//   npx vitest run tests/review/findings-RED.test.ts
import { describe, it, expect } from 'vitest';
import { linesReadFrom, specFromText } from '@/lib/peak-order/spec-from-text';

/** These lines (0-based) must stay in the notes. */
const stays = (text: string, ...lines: number[]): void => {
  const dropped = linesReadFrom(text);
  for (const n of lines) expect(dropped, `line ${n} ${JSON.stringify(text.split('\n')[n])} left the notes`).not.toContain(n);
};
const fields = (text: string): Record<string, unknown> => specFromText(text).fields as Record<string, unknown>;

  it('"เนื้อในและปก …"', () => stays('วารสาร A4 16 หน้า\nเนื้อในและปก กระดาษอาร์ตมัน 120 แกรม\nพิมพ์ 4 สี', 2));
  it('"เนื้อในและปก" alone as a heading', () => stays('วารสาร A4 16 หน้า\nเนื้อในและปก\nกระดาษอาร์ตมัน 120 แกรม\nพิมพ์ 4 สี', 2, 3));
